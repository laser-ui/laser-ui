# en-US

Use the `placement` prop to set the popup position.

# zh-CN

使用 `placement` 属性设置弹出位置。

```tsx
import { Compose, Radio, Select, Separator } from '@laser-ui/components';
import { useState } from 'react';

const list = Array.from({ length: 100 }).map((_, index) => ({
  label: `item ${index}`,
  value: index,
  disabled: index === 3,
}));
export default function Demo() {
  const [placement, setPlacement] = useState('bottom-left');

  return (
    <>
      <Radio.Group
        list={['top-left', 'top', 'top-right', 'bottom-left', 'bottom', 'bottom-right'].map((placement) => ({
          label: placement,
          value: placement,
        }))}
        model={placement}
        pattern="outline"
        onModelChange={(value) => {
          setPlacement(value);
        }}
      >
        {(groupProps, optionProps, options) => (
          <Compose {...groupProps}>
            {options.map((opt) => (
              <Radio key={opt.value} {...optionProps(opt)} />
            ))}
          </Compose>
        )}
      </Radio.Group>
      <Separator />
      <div className="row g-2">
        <div className="col-auto">
          <Select list={list} placeholder="Placement" placement={placement} />
        </div>
        <div className="col-auto">
          <Select list={list} placeholder="monospaced:false" placement={placement} monospaced={false} />
        </div>
      </div>
    </>
  );
}
```
